        val error = runCatching {
            response()?.errorBody()?.string()?.let {
                Gson().fromJson(it, ErrorResponseDto::class.java)
            }
        }.getOrNull()

        val detalles = error?.errores
            ?.values
            ?.filter { it.isNotBlank() }
            ?.joinToString("\n")

        if (!detalles.isNullOrBlank()) {
            return detalles
        }

        error?.mensaje?.takeIf { it.isNotBlank() }?.let {
            return it
        }

            400 -> "Los datos ingresados no son vÃ¡lidos"

            409 -> "Ya existe una candidata con ese correo"
            in 500..599 -> "El servidor presenta un problema"

object CalculadoraPuntaje {
    const val PUNTAJE_MINIMO = 1
    const val PUNTAJE_MAXIMO = 5

    fun calcular(criterios: List<Criterio>, puntajes: Map<Long, Int>): Double {
        var pesoTotal = 0
        var obtenido = 0

        // Recorre cada criterio uno por uno
        for (criterio in criterios) {
            pesoTotal += criterio.peso

            // Busca la nota, y si no existe se usa 0 por defecto
            val nota = puntajes[criterio.id] ?: 0
            obtenido += nota * criterio.peso
        }

        if (pesoTotal == 0) return 0.0
        return obtenido.toDouble() / (pesoTotal * PUNTAJE_MAXIMO) * 100.0
    }

    fun categoria(puntaje: Double): String = when {
        puntaje >= 85 -> "Sobresaliente"
        puntaje >= 70 -> "SÃ³lida"
        puntaje >= 50 -> "En desarrollo"
        puntaje > 0 -> "Inicial"
        else -> "Sin puntaje"
    }

    fun nivel(puntaje: Int?): String = when (puntaje) {
        1 -> "Insuficiente"
        2 -> "BÃ¡sico"
        3 -> "Adecuado"
        4 -> "Destacado"
        5 -> "Sobresaliente"
        else -> "Sin asignar"

import android.content.Context
import android.graphics.pdf.PdfRenderer
import android.net.Uri
import android.os.ParcelFileDescriptor
import android.provider.OpenableColumns
import java.io.File
import java.util.UUID
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext

data class CvImportado(val uri: String, val nombre: String, val tamano: Long)

// Copia el documento seleccionado al almacenamiento privado para conservarlo al reiniciar
object ArchivoPostulacion {
    suspend fun importar(context: Context, uri: Uri): CvImportado = withContext(Dispatchers.IO) {
        val limite = 5L * 1024 * 1024
        val nombre = context.contentResolver.query(uri, arrayOf(OpenableColumns.DISPLAY_NAME), null, null, null)?.use {
            if (it.moveToFirst()) it.getString(0) else null
        } ?: "curriculum.pdf"
        require(nombre.endsWith(".pdf", ignoreCase = true)) { "Selecciona un archivo PDF" }
        val carpeta = File(context.filesDir, "cvs").apply { mkdirs() }
        val destino = File(carpeta, "${UUID.randomUUID()}.pdf")
        try {
            val entrada = context.contentResolver.openInputStream(uri) ?: error("No se pudo abrir el documento")
            var total = 0L
            entrada.use { input ->
                destino.outputStream().use { output ->
                    val buffer = ByteArray(8192)
                    while (true) {
                        val cantidad = input.read(buffer)
                        if (cantidad < 0) break
                        total += cantidad
                        require(total <= limite) { "El PDF supera los 5 MB" }
                        output.write(buffer, 0, cantidad)
                    }
                }
            }
            require(total > 0) { "El documento estÃ¡ vacÃ­o" }
            ParcelFileDescriptor.open(destino, ParcelFileDescriptor.MODE_READ_ONLY).use { descriptor ->
                PdfRenderer(descriptor).use { pdf -> require(pdf.pageCount > 0) { "PDF sin pÃ¡ginas" } }
            }
            CvImportado(Uri.fromFile(destino).toString(), nombre, total)
        } catch (e: Exception) {
            destino.delete()
            throw e
        }
    }
}

import android.os.Build
import androidx.biometric.BiometricManager
import androidx.biometric.BiometricManager.Authenticators.BIOMETRIC_WEAK
import androidx.biometric.BiometricManager.Authenticators.DEVICE_CREDENTIAL
import android.content.Context
import androidx.biometric.BiometricPrompt
import androidx.core.content.ContextCompat
import androidx.fragment.app.FragmentActivity

object BiometriaHelper {

    private fun autenticadores(): Int {
        if (Build.VERSION.SDK_INT >= 30) return BIOMETRIC_WEAK or DEVICE_CREDENTIAL
        return BIOMETRIC_WEAK
    }

    fun problema(context: Context): String? {
        return when (BiometricManager.from(context).canAuthenticate(autenticadores())) {
            BiometricManager.BIOMETRIC_SUCCESS -> null
            BiometricManager.BIOMETRIC_ERROR_NONE_ENROLLED -> "Configura una huella o bloqueo en los ajustes de seguridad"
            else -> "La verificaciÃ³n no estÃ¡ disponible en este dispositivo"
        }
    }

    fun autenticar(context: Context, alExito: () -> Unit, alError: (String) -> Unit) {
        val activity = context as? FragmentActivity
        if (activity == null) {
            alError("El contexto no es vÃ¡lido para biometrÃ­a")
            return
        }

        val error = problema(context)
        if (error != null) {
            alError(error)
            return
        }

        val executor = ContextCompat.getMainExecutor(context)
        val biometricPrompt = BiometricPrompt(activity, executor,
            object : BiometricPrompt.AuthenticationCallback() {
                // Si la huella es correcta, ejecutamos la acciÃ³n de Ã©xito
                override fun onAuthenticationSucceeded(result: BiometricPrompt.AuthenticationResult) {
                    alExito()
                }
                override fun onAuthenticationError(errorCode: Int, errString: CharSequence) {
                    alError(errString.toString())
                }
            })

        val builder = BiometricPrompt.PromptInfo.Builder()
            .setTitle("Acceso Restringido")
            .setSubtitle(if (Build.VERSION.SDK_INT >= 30) "Usa tu huella o bloqueo para acceder a DirecciÃ³n" else "Usa tu biometrÃ­a para acceder a DirecciÃ³n")
            .setAllowedAuthenticators(autenticadores())
        if (Build.VERSION.SDK_INT < 30) builder.setNegativeButtonText("Cancelar")
        val promptInfo = builder.build()

        biometricPrompt.authenticate(promptInfo)
    }
}

    }

            estado.nombre,
            estado.correo,

        )

                    null

                }
            )
            return estado.copy(errores = erroresCompletos, formularioValido = !erroresCompletos.hayErrores)
        }

        return estado.copy(errores = errores, formularioValido = !errores.hayErrores)
    }

                    registradaPor = original?.registradaPor ?: responsable
                )
                val idGenerado = if (original == null) repository.registrarEnServidor(candidata)
                else {

                }

                _uiState.value = _uiState.value.copy(guardando = false, errorGuardado = e.mensajeParaUsuario())
            }
        }
    }

class SesionViewModel(
    private val historial: HistorialRepository,
    private val auth: AuthRepository

    private val _sesion = MutableStateFlow<Sesion?>(null)
    val sesion = _sesion.asStateFlow()
    private val _login = MutableStateFlow(LoginUiState())
    val loginState = _login.asStateFlow()
    private var pendiente: Sesion? = null
    private var acceso: Job? = null

    fun onUsuario(valor: String) {
        if (_login.value.cargando) return
        _login.update { it.copy(usuario = valor, errorUsuario = null, error = null) }
    }

    fun onPassword(valor: String) {
        if (_login.value.cargando) return
        _login.update { it.copy(password = valor, errorPassword = null, error = null) }
    }

    fun iniciarSesion() {
        if (_login.value.cargando || _sesion.value != null) return
        val datos = _login.value
        val usuarioError = if (datos.usuario.isBlank()) "Ingresa tu usuario" else null
        val passwordError = if (datos.password.isBlank()) "Ingresa tu contraseÃ±a" else null
        _login.update { it.copy(errorUsuario = usuarioError, errorPassword = passwordError, error = null) }
        if (usuarioError != null || passwordError != null) return
        _login.update { it.copy(cargando = true) }
        acceso = viewModelScope.launch {
            try {
                val actual = auth.iniciarSesion(datos.usuario.trim(), datos.password)
                if (actual.rol == Rol.DIRECCION) {
                    pendiente = actual
                    _login.update { it.copy(password = "", requiereBiometria = true) }
                } else completarAcceso(actual)
            } catch (e: CancellationException) { throw e }
            catch (e: Exception) {
                pendiente = null
                _login.update { it.copy(cargando = false, error = e.mensajeParaUsuario()) }
            }
        }
    }

    fun confirmarBiometria() {
        val actual = pendiente ?: return
        pendiente = null
        _login.update { it.copy(requiereBiometria = false) }
        acceso = viewModelScope.launch { completarAcceso(actual) }
    }

    fun cancelarBiometria(mensaje: String) {
        if (pendiente == null) return
        pendiente = null
        _login.update { it.copy(cargando = false, requiereBiometria = false, error = mensaje) }
    }

    private suspend fun completarAcceso(actual: Sesion) {
        try {
            historial.registrar(null, Acciones.INICIO_SESION, "Ingreso como ${actual.rol.etiqueta}", actual.nombre)
            _login.value = LoginUiState()
            _sesion.value = actual
        } catch (e: CancellationException) { throw e }
        catch (e: Exception) {
            _login.update { it.copy(cargando = false, requiereBiometria = false,
                error = "No se pudo registrar el acceso. Intenta nuevamente.") }
        }
    }

    fun cerrar() {
        acceso?.cancel()
        pendiente = null
        val actual = _sesion.value
        _sesion.value = null
        _login.value = LoginUiState()
        if (actual != null) viewModelScope.launch {
            try {
                historial.registrar(null, Acciones.CIERRE_SESION, "Cierre de sesiÃ³n", actual.nombre)
            } catch (e: CancellationException) { throw e }
            catch (e: Exception) {

            }
        }
    }
